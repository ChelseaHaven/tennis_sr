cmake_minimum_required(VERSION 3.16)
project(mecanum_gz)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(mecanum_force_system SHARED src/force_move.cpp)
target_compile_features(mecanum_force_system PUBLIC cxx_std_17)
target_link_libraries(mecanum_force_system PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS mecanum_force_system LIBRARY DESTINATION lib)
ament_package()
